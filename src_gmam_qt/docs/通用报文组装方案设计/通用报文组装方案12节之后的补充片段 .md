```cpp
#pragma once
#include <QByteArray>
#include <QHash>
#include <QString>
#include <QVariant>

// 字段类型
enum class FieldType {
    Numeric,    // 数值型 N
    Chars,      // 字符型 C
    Reserved,   // 预留，填空格
};
// 小数处理方式（仅 Numeric 用）
enum class Rounding {
    Truncate,   // 小数位截断（默认）
    Round,      // 四舍五入
};
// 超额策略（编码后长度 > length；规则 7 可配）
enum class OverflowPolicy {
    Error,      // 超长 → 返回空 + error
    Truncate,   // 超长 → 按类型对齐方向截断（兼容旧 memcpy）
    FillChar,   // 超长 → 整字段填 overflowFill
};
// 单个字段描述
struct FieldDetail {
    QString   name;
    FieldType type     = FieldType::Reserved;
    int       length   = 0;                    // 字节数（占位）
    int       scale    = 0;                    // 10^-scale，仅 Numeric 用
    Rounding  rounding = Rounding::Truncate;   // 仅 Numeric 用（小数）
    OverflowPolicy overflow = OverflowPolicy::Error;
    char      overflowFill = '9';             // 仅 FillChar 时使用
};
// 数据内容组装器
class DataContentBuilder {
public:
    // maxLength: 数据内容最大字节数，默认 8162

    // 数据内容最大长度
    int maxLength() const { return m_maxLength; }

    QVector<FieldDetail> m_schema;
    int m_totalLength = 0;
```

```
## 三、替换第 6.2 节 `DataContentBuilder.cpp` 中的构造与 build

只改这两处，其余（`parseNumericString`、`encodeField`）不变。

```
```cpp
// ---------------------------------------------------------
// 构造 / 组装
// ---------------------------------------------------------
DataContentBuilder::DataContentBuilder(QVector<FieldDetail> schema,
                                       int maxLength)
    : m_schema(std::move(schema))
    , m_maxLength(maxLength)
{
    for (const auto& f : m_schema) {
        if (f.length > 0)
            m_totalLength += f.length;
    }
}
QByteArray DataContentBuilder::build(
        const QHash<QString, QString>& values,
        QString& error) const
{
    error.clear();
    // 数据区总长校验
    if (m_totalLength > m_maxLength) {
        error = QString("数据内容超长：%1 > %2")
                    .arg(m_totalLength).arg(m_maxLength);
        return {};
    }
    QByteArray buffer;
    buffer.reserve(m_totalLength);
    for (const FieldDetail& f : m_schema) {
        if (f.length <= 0) {
            error = QString("字段 %1 长度非法").arg(f.name);
            return {};
        }
        // Reserved：不取值，直接填空格
        if (f.type == FieldType::Reserved) {
            buffer.append(f.length, ' ');
            continue;
        }
        // 取值；缺值 = 无法采集
        auto it = values.find(f.name);
        if (it == values.end()) {
            if (f.type == FieldType::Numeric)
                buffer.append(f.length, '0');
            else
                buffer.append(f.length, ' ');
            continue;
        }
        QByteArray fieldBytes;
        if (!encodeField(it.value(), f, fieldBytes, error))
            return {};
        buffer.append(fieldBytes);
    }
    return buffer;
}
```
